#!/bin/bash
set -e # Exit immediately if any command fails

export PYTHONPATH="/home/jovyan/work/BMO-Project/personaplex_repo/moshi"
# Isolating to the second GPU (index 1)
export CUDA_VISIBLE_DEVICES=1

echo "=== Starting 4-Tier Sweep (Layers 0-30) on GPU 2 ==="

# --- Config A: The Safe Cushion ---
# 0.1% FP16 / 1% INT8 / 10% INT4 / 88.9% INT2
echo "Running Config A (0.1% FP16, 1% INT8, 10% INT4, 88.9% INT2)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 2 \
  --ratio-fp16 0.001 \
  --ratio-int8 0.01 \
  --ratio-int4 0.10 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 0-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_4tier_A.pt 2>&1 | tee 4tier_A.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_4tier_A.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_4tier_A.json 2>&1 | tee zs_4tier_A.log

# --- Config B: The Aggressive Squeeze ---
# 0.1% FP16 / 0.5% INT8 / 5% INT4 / 94.4% INT2
echo "Running Config B (0.1% FP16, 0.5% INT8, 5% INT4, 94.4% INT2)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 2 \
  --ratio-fp16 0.001 \
  --ratio-int8 0.005 \
  --ratio-int4 0.05 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 0-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_4tier_B.pt 2>&1 | tee 4tier_B.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_4tier_B.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_4tier_B.json 2>&1 | tee zs_4tier_B.log

# --- Config C: The Heavy Brain ---
# 0.5% FP16 / 2% INT8 / 15% INT4 / 82.5% INT2
echo "Running Config C (0.5% FP16, 2% INT8, 15% INT4, 82.5% INT2)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 2 \
  --ratio-fp16 0.005 \
  --ratio-int8 0.02 \
  --ratio-int4 0.15 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 0-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_4tier_C.pt 2>&1 | tee 4tier_C.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_4tier_C.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_4tier_C.json 2>&1 | tee zs_4tier_C.log

# --- Ranking ---
echo "=== Ranking Results ==="
python -u rank_multitier_runs.py \
  --run A:bmo_temporal_4tier_A.pt:zs_4tier_A.json \
  --run B:bmo_temporal_4tier_B.pt:zs_4tier_B.json \
  --run C:bmo_temporal_4tier_C.pt:zs_4tier_C.json \
  --min-median-cos 0.997 \
  --max-weight-gib 3.6
