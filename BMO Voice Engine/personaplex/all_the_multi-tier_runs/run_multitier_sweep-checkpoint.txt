#!/bin/bash
set -e # Exit immediately if any command fails

export PYTHONPATH="/home/jovyan/work/BMO-Project/personaplex_repo/moshi"
# Isolating to the second GPU (index 1)
export CUDA_VISIBLE_DEVICES=1

echo "=== Starting Multi-Tier Sweep on GPU 2 ==="

# --- Config A ---
echo "Running Config A (0.1% FP16, 1% INT8, 98.9% INT3)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 3 \
  --ratio-fp16 0.001 \
  --ratio-int8 0.01 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 16-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_multitier_A.pt 2>&1 | tee multitier_A.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_multitier_A.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_A.json 2>&1 | tee zs_A.log

# --- Config B ---
echo "Running Config B (0.5% FP16, 2% INT8, 97.5% INT2)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 2 \
  --ratio-fp16 0.005 \
  --ratio-int8 0.02 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 16-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_multitier_B.pt 2>&1 | tee multitier_B.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_multitier_B.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_B.json 2>&1 | tee zs_B.log

# --- Config C ---
echo "Running Config C (0.1% FP16, 5% INT8, 94.9% INT3)..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 3 \
  --ratio-fp16 0.001 \
  --ratio-int8 0.05 \
  --block-size 128 \
  --max-calibration-samples 16384 --max-steps-per-clip 750 --max-clips 857 \
  --quantize-layers 16-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_multitier_C.pt 2>&1 | tee multitier_C.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_multitier_C.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_C.json 2>&1 | tee zs_C.log

# --- Ranking ---
echo "=== Ranking Results ==="
python -u rank_multitier_runs.py \
  --run A:bmo_temporal_multitier_A.pt:zs_A.json \
  --run B:bmo_temporal_multitier_B.pt:zs_B.json \
  --run C:bmo_temporal_multitier_C.pt:zs_C.json \
  --min-median-cos 0.997 \
  --max-weight-gib 3.6
