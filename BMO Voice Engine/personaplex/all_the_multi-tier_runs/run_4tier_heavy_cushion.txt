#!/bin/bash
set -e

export PYTHONPATH="/home/jovyan/work/BMO-Project/personaplex_repo/moshi"
export CUDA_VISIBLE_DEVICES=1

echo "=== Starting 4-Tier HEAVY CUSHION (Layers 0-30) on GPU 2 ==="

# 2% FP16 / 8% INT8 / 20% INT4 / 70% INT2
# Target BPW: ~5.16 | Target Size: ~3.55 GB
echo "Running Heavy Cushion..."
python -u apply_septq_multitier.py \
  --device cuda:0 \
  --bf16 v5_step1500_split.safetensors \
  --calibration-clips bmo_dataset_clean \
  --bits 2 \
  --ratio-fp16 0.02 \
  --ratio-int8 0.08 \
  --ratio-int4 0.20 \
  --block-size 128 \
  --max-calibration-samples 8192 \
  --max-steps-per-clip 750 \
  --max-clips 400 \
  --quantize-layers 0-30 \
  --skip-modules "self_attn.out_proj" \
  --out bmo_temporal_4tier_heavy.pt 2>&1 | tee 4tier_heavy.log

python -u verify_septq_zs_drift.py \
  --device cuda:0 \
  --teacher v5_step1500_split.safetensors \
  --student bmo_temporal_4tier_heavy.pt \
  --steps 125 \
  --min-median-cos 0.997 \
  --save-json zs_4tier_heavy.json 2>&1 | tee zs_4tier_heavy.log
